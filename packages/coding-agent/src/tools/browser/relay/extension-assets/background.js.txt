// extension/debugger-ownership.ts
async function ownedDebuggerTabs(targets, probe) {
  const owned = await Promise.all(targets.map(async (target) => {
    if (!target.attached || target.tabId === undefined)
      return;
    try {
      await probe(target.tabId);
      return target.tabId;
    } catch {
      return;
    }
  }));
  return owned.filter((tabId) => tabId !== undefined);
}

// extension/download-pairing.ts
var DOWNLOAD_PAIRING_MS = 5000;
var DOWNLOAD_LOOKUP_MS = 60000;

class DownloadPairing {
  #now;
  #pairs = new Map;
  #reports = [];
  constructor(now = Date.now) {
    this.#now = now;
  }
  report(guid, url) {
    if (this.#pairs.has(guid) || this.#reports.some((report) => report.guid === guid))
      return;
    this.#dropAgedReports();
    this.#reports.push({ guid, url, at: this.#now() });
  }
  created(id, url) {
    this.#dropAgedReports();
    const index = this.#reports.findIndex((report) => report.url === url);
    if (index < 0)
      return;
    this.#pairs.set(this.#reports[index].guid, { id });
    this.#reports.splice(index, 1);
  }
  #dropAgedReports() {
    const oldest = this.#now() - DOWNLOAD_PAIRING_MS;
    while (this.#reports.length > 0 && this.#reports[0].at < oldest)
      this.#reports.shift();
  }
  finished(id, state) {
    const now = this.#now();
    for (const [guid, pair] of this.#pairs) {
      if (pair.id === id) {
        if (state === "interrupted")
          this.#pairs.delete(guid);
        else
          pair.completedAt = now;
      } else if (pair.completedAt !== undefined && pair.completedAt < now - DOWNLOAD_LOOKUP_MS) {
        this.#pairs.delete(guid);
      }
    }
  }
  idFor(guid) {
    return this.#pairs.get(guid)?.id;
  }
  forget(guid) {
    this.#pairs.delete(guid);
  }
}

// extension/foreign-frames.ts
var RETRY_DELAYS_MS = [50, 100, 200, 400];
var sweeps = new Map;
async function pastForeignFrames(tabId, request) {
  for (let attempt = 0;; attempt++) {
    try {
      return await request();
    } catch (error) {
      if (attempt === RETRY_DELAYS_MS.length || !FOREIGN_FRAME_REFUSAL.test(String(error)) || await sweep(tabId) === 0)
        throw error;
    }
    const { promise, resolve } = Promise.withResolvers();
    setTimeout(resolve, RETRY_DELAYS_MS[attempt]);
    await promise;
  }
}
function sweep(tabId) {
  let pending = sweeps.get(tabId);
  if (!pending) {
    pending = neutralizeForeignFrames(tabId).finally(() => sweeps.delete(tabId));
    sweeps.set(tabId, pending);
  }
  return pending;
}
var reportHidden = () => {};
function onForeignFramesHidden(report) {
  reportHidden = report;
}
var MAX_PATH_LENGTH = 100;
var MAX_TITLE_LENGTH = 60;
var PROBE_TIMEOUT_MS = 500;
var PROBE_MAX_BYTES = 64 * 1024;
async function reportEmptied(tabId, srcsByFrame) {
  const probes = await Promise.allSettled(Array.from(srcsByFrame, ([frameId, srcs]) => chrome.scripting.executeScript({
    target: { tabId, frameIds: [frameId] },
    world: "ISOLATED",
    injectImmediately: true,
    args: [srcs, PROBE_TIMEOUT_MS, PROBE_MAX_BYTES],
    func: async (urls, timeoutMs, maxBytes) => await Promise.all(urls.map(async (url) => {
      try {
        const response = await fetch(url, { signal: AbortSignal.timeout(timeoutMs) });
        if (!response.ok)
          return null;
        const reader = response.body?.getReader();
        const decoder = new TextDecoder;
        let head = "";
        for (let read = 0;reader && read < maxBytes && !/<\/title>/i.test(head); ) {
          const chunk = await reader.read();
          if (chunk.done)
            break;
          read += chunk.value.byteLength;
          head += decoder.decode(chunk.value, { stream: true });
        }
        reader?.cancel().catch(() => {});
        const title = /<title[^>]*>([^<]*)<\/title>/i.exec(head.slice(0, maxBytes))?.[1]?.trim();
        return { url: response.url, title };
      } catch {
        return null;
      }
    }))
  })));
  const frames = new Map;
  for (const probe of probes) {
    if (probe.status === "rejected")
      continue;
    for (const loaded of probe.value.flatMap(({ result }) => result ?? [])) {
      if (!loaded?.url.startsWith("chrome-extension://"))
        continue;
      const { host: extensionId, pathname } = new URL(loaded.url);
      if (!/^[a-p]{32}$/.test(extensionId) || extensionId === chrome.runtime.id)
        continue;
      const path = pathname.slice(1, 1 + MAX_PATH_LENGTH);
      const title = loaded.title?.slice(0, MAX_TITLE_LENGTH);
      frames.set(`${extensionId}/${path}`, { extensionId, path, ...title ? { title } : {} });
    }
  }
  if (frames.size > 0)
    reportHidden(tabId, [...frames.values()]);
}
async function neutralizeForeignFrames(tabId) {
  let results;
  try {
    results = await chrome.scripting.executeScript({
      target: { tabId, allFrames: true },
      injectImmediately: true,
      args: [chrome.runtime.id],
      func: (self) => {
        const swept = { found: 0, emptied: [] };
        const visit = (root) => {
          const walker = document.createTreeWalker(root, NodeFilter.SHOW_ELEMENT);
          for (let element = walker.nextNode();element; element = walker.nextNode()) {
            const tag = element.tagName.toUpperCase();
            if (tag === "IFRAME" || tag === "FRAME") {
              const src = element.getAttribute("src") ?? "";
              if (src.startsWith("chrome-extension://") && !src.startsWith(`chrome-extension://${self}/`)) {
                if (tag === "FRAME") {
                  element.setAttribute("src", "about:blank");
                  swept.emptied.push(src);
                } else if (element.getAttribute("srcdoc") === null) {
                  element.setAttribute("srcdoc", "");
                  swept.emptied.push(src);
                }
                element.style.setProperty("display", "none", "important");
                swept.found++;
              }
            }
            const shadow = chrome.dom.openOrClosedShadowRoot(element);
            if (shadow)
              visit(shadow);
          }
        };
        visit(document);
        return swept;
      }
    });
  } catch {
    return 0;
  }
  let found = 0;
  const srcsByFrame = new Map;
  for (const { frameId, result } of results) {
    found += result?.found ?? 0;
    if (result && result.emptied.length > 0)
      srcsByFrame.set(frameId, result.emptied);
  }
  if (srcsByFrame.size > 0)
    reportEmptied(tabId, srcsByFrame).catch(() => {});
  return found;
}

// extension/policy.ts
var APPROVAL_TTL_MS = 5 * 60000;
var ONE_TIME_GRANT_TTL_MS = 5 * 60000;
var BLOCKED_SCHEME_RE = /^(chrome|chrome-extension|devtools|edge|view-source|chrome-untrusted|chrome-search):/i;
var SENSITIVE_CDP_METHODS = new Set([
  "Storage.getCookies",
  "Network.getAllCookies",
  "Network.getCookies",
  "Network.requestWillBeSentExtraInfo",
  "Network.responseReceivedExtraInfo"
]);
function isBlockedUrl(raw) {
  const u = raw.trim();
  if (!u)
    return false;
  if (BLOCKED_SCHEME_RE.test(u))
    return true;
  if (/^about:/i.test(u) && u.toLowerCase() !== "about:blank")
    return true;
  return false;
}
function isFileUrl(raw) {
  return /^file:/i.test(raw.trim());
}
function parseHostname(raw) {
  try {
    return new URL(raw).hostname.toLowerCase();
  } catch {
    return "";
  }
}
function registrableDomain(rawUrl) {
  const host = parseHostname(rawUrl);
  if (!host)
    return "";
  if (host === "localhost" || /^\d+\.\d+\.\d+\.\d+$/.test(host) || host.includes(":")) {
    return host;
  }
  const parts = host.split(".").filter(Boolean);
  if (parts.length <= 2)
    return host;
  const lastTwo = parts.slice(-2).join(".");
  const secondLevelPublic = new Set(["co.uk", "com.cn", "org.cn", "gov.cn", "net.cn", "com.hk", "com.tw", "com.au", "co.jp"]);
  if (secondLevelPublic.has(lastTwo) && parts.length >= 3) {
    return parts.slice(-3).join(".");
  }
  return lastTwo;
}
function normalizeUrlForFingerprint(raw) {
  const trimmed = raw.trim();
  if (!trimmed)
    return "about:blank";
  try {
    const u = new URL(trimmed);
    u.hash = "";
    return u.toString();
  } catch {
    return trimmed;
  }
}

class CloudPolicyEngine {
  #sessions = new Map;
  #seq = 0;
  #ensure(threadId) {
    let s = this.#sessions.get(threadId);
    if (!s) {
      s = {
        threadId,
        sharedTabs: new Map,
        autoApproveCreateTab: false,
        autoApproveCategories: new Set,
        oneTimeGrants: new Map,
        pendingApprovals: new Map
      };
      this.#sessions.set(threadId, s);
    }
    return s;
  }
  #pruneExpired(s, now) {
    for (const [fp, g] of s.oneTimeGrants) {
      if (g.expiresAt <= now)
        s.oneTimeGrants.delete(fp);
    }
    for (const [id, p] of s.pendingApprovals) {
      if (p.expiresAt <= now)
        s.pendingApprovals.delete(id);
    }
  }
  shareTab(threadId, snap) {
    const s = this.#ensure(threadId);
    const rec = {
      tabId: snap.tabId,
      source: "shared",
      initialHost: registrableDomain(snap.url),
      url: snap.url,
      title: snap.title
    };
    s.sharedTabs.set(snap.tabId, rec);
    return rec;
  }
  registerCreatedTab(threadId, snap) {
    const s = this.#ensure(threadId);
    const rec = {
      tabId: snap.tabId,
      source: "created",
      initialHost: registrableDomain(snap.url),
      url: snap.url,
      title: snap.title
    };
    s.sharedTabs.set(snap.tabId, rec);
    return rec;
  }
  unshareTab(threadId, tabId) {
    const s = this.#sessions.get(threadId);
    if (!s)
      return false;
    return s.sharedTabs.delete(tabId);
  }
  updateKnownTab(threadId, snap) {
    const s = this.#sessions.get(threadId);
    const rec = s?.sharedTabs.get(snap.tabId);
    if (!rec)
      return false;
    rec.url = snap.url;
    rec.title = snap.title;
    if (!rec.initialHost && snap.url) {
      rec.initialHost = registrableDomain(snap.url);
    }
    return true;
  }
  sessionsHoldingTab(tabId) {
    const out = [];
    for (const [tid, s] of this.#sessions) {
      if (s.sharedTabs.has(tabId))
        out.push(tid);
    }
    return out;
  }
  clearSession(threadId) {
    const s = this.#sessions.get(threadId);
    if (!s)
      return [];
    const tabs = [...s.sharedTabs.keys()];
    this.#sessions.delete(threadId);
    return tabs;
  }
  clearAll() {
    const all = new Set;
    for (const s of this.#sessions.values()) {
      for (const id of s.sharedTabs.keys())
        all.add(id);
    }
    this.#sessions.clear();
    return [...all];
  }
  filterHello(threadId, fullHello) {
    const s = this.#sessions.get(threadId);
    if (!s || s.sharedTabs.size === 0) {
      return {
        ...fullHello,
        downloadsProtocol: undefined,
        tabs: [],
        attachedTabIds: []
      };
    }
    const tabs = fullHello.tabs.filter((tab) => s.sharedTabs.has(tab.tabId));
    const attachedTabIds = fullHello.attachedTabIds.filter((id) => s.sharedTabs.has(id));
    return {
      ...fullHello,
      downloadsProtocol: undefined,
      tabs,
      attachedTabIds
    };
  }
  filterTabEvent(threadId, ev) {
    const s = this.#sessions.get(threadId);
    if (!s)
      return null;
    const tabId = ev.t === "tabRemoved" ? ev.tabId : ev.tab.tabId;
    if (!s.sharedTabs.has(tabId))
      return null;
    if (ev.t !== "tabRemoved") {
      this.updateKnownTab(threadId, ev.tab);
    }
    return ev;
  }
  authorizeRpc(threadId, rpc, now = Date.now()) {
    const s = this.#ensure(threadId);
    this.#pruneExpired(s, now);
    if (rpc.op === "findDownload") {
      return { kind: "deny", reason: "Oryn cloud relay disables findDownload to avoid exposing local file paths." };
    }
    if (rpc.op === "createTab") {
      const url = rpc.url ?? "";
      if (isBlockedUrl(url)) {
        return { kind: "deny", reason: `Blocked internal browser URL: ${url}` };
      }
      const isFile = isFileUrl(url);
      const category = isFile ? "file_url" : "create_tab";
      const fp = `${category}:${normalizeUrlForFingerprint(url)}`;
      if (!isFile && s.autoApproveCreateTab)
        return { kind: "allow" };
      if (isFile && s.autoApproveCategories.has("file_url"))
        return { kind: "allow" };
      if (this.#consumeOneTimeGrant(s, fp, now))
        return { kind: "allow" };
      const summary = isFile ? `打开本地文件页面 ${url}` : `新建标签页 ${url || "about:blank"}`;
      return this.#requireApproval(s, { category, fingerprint: fp, summary, url }, now);
    }
    if (rpc.op === "group" || rpc.op === "ungroup") {
      for (const id of rpc.tabIds) {
        if (!s.sharedTabs.has(id)) {
          return { kind: "deny", reason: `Tab ${id} is not shared with this Oryn thread.` };
        }
      }
      return { kind: "allow" };
    }
    const tabRec = s.sharedTabs.get(rpc.tabId);
    if (!tabRec) {
      return {
        kind: "deny",
        reason: `Tab ${rpc.tabId} is not shared with this Oryn thread. Ask the user to share it in the Chrome extension popup.`
      };
    }
    if (rpc.op !== "send") {
      return { kind: "allow" };
    }
    const method = rpc.method;
    if (method.startsWith("Target.") || method.startsWith("Browser.") || method.startsWith("SystemInfo.")) {
      const category = "browser_target_cdp";
      const fp = `${category}:${rpc.tabId}:${method}`;
      if (s.autoApproveCategories.has(category) || this.#consumeOneTimeGrant(s, fp, now)) {
        return { kind: "allow" };
      }
      return this.#requireApproval(s, { category, fingerprint: fp, summary: `执行浏览器级 CDP 命令 ${method}`, tabId: rpc.tabId, method }, now);
    }
    if (SENSITIVE_CDP_METHODS.has(method)) {
      const category = "sensitive_cdp";
      const fp = `${category}:${rpc.tabId}:${method}`;
      if (s.autoApproveCategories.has(category) || this.#consumeOneTimeGrant(s, fp, now)) {
        return { kind: "allow" };
      }
      return this.#requireApproval(s, { category, fingerprint: fp, summary: `读取标签页凭据/请求头（${method}）`, tabId: rpc.tabId, method }, now);
    }
    if (method === "Page.navigate") {
      const targetUrl = typeof rpc.params?.url === "string" ? rpc.params.url : "";
      if (isBlockedUrl(targetUrl)) {
        return { kind: "deny", reason: `Blocked internal browser URL: ${targetUrl}` };
      }
      if (isFileUrl(targetUrl)) {
        const category = "file_url";
        const fp = `${category}:${rpc.tabId}:${normalizeUrlForFingerprint(targetUrl)}`;
        if (s.autoApproveCategories.has(category) || this.#consumeOneTimeGrant(s, fp, now)) {
          return { kind: "allow" };
        }
        return this.#requireApproval(s, { category, fingerprint: fp, summary: `导航到本地文件 ${targetUrl}`, tabId: rpc.tabId, url: targetUrl, method }, now);
      }
      if (tabRec.source === "shared" && targetUrl) {
        const nextDomain = registrableDomain(targetUrl);
        if (tabRec.initialHost && nextDomain && nextDomain !== tabRec.initialHost) {
          const category = "cross_origin_nav";
          const fp = `${category}:${rpc.tabId}:${nextDomain}`;
          if (s.autoApproveCategories.has(category) || this.#consumeOneTimeGrant(s, fp, now)) {
            return { kind: "allow" };
          }
          return this.#requireApproval(s, {
            category,
            fingerprint: fp,
            summary: `共享标签页跨站跳转到 ${nextDomain}（原域名 ${tabRec.initialHost}）`,
            tabId: rpc.tabId,
            url: targetUrl,
            method
          }, now);
        }
      }
    }
    return { kind: "allow" };
  }
  resolveApproval(threadId, approvalId, decision, now = Date.now()) {
    const s = this.#sessions.get(threadId);
    if (!s)
      return null;
    this.#pruneExpired(s, now);
    const approval = s.pendingApprovals.get(approvalId);
    if (!approval)
      return null;
    s.pendingApprovals.delete(approvalId);
    if (decision === "approved_once") {
      s.oneTimeGrants.set(approval.fingerprint, { expiresAt: now + ONE_TIME_GRANT_TTL_MS });
    } else if (decision === "approved_session") {
      if (approval.category === "create_tab") {
        s.autoApproveCreateTab = true;
      } else {
        s.autoApproveCategories.add(approval.category);
      }
      for (const [id, p] of s.pendingApprovals) {
        if (p.category === approval.category)
          s.pendingApprovals.delete(id);
      }
    }
    return { approval, decision };
  }
  setAutoApproveCreateTab(threadId, enabled) {
    const s = this.#ensure(threadId);
    s.autoApproveCreateTab = enabled;
  }
  getSummary(threadId, now = Date.now()) {
    const s = this.#sessions.get(threadId);
    if (!s) {
      return {
        threadId,
        sharedTabs: [],
        pendingApprovals: [],
        autoApproveCreateTab: false,
        autoApproveCategories: []
      };
    }
    this.#pruneExpired(s, now);
    return {
      threadId,
      sharedTabs: [...s.sharedTabs.values()],
      pendingApprovals: [...s.pendingApprovals.values()],
      autoApproveCreateTab: s.autoApproveCreateTab,
      autoApproveCategories: [...s.autoApproveCategories]
    };
  }
  totalPendingApprovals(now = Date.now()) {
    let total = 0;
    for (const s of this.#sessions.values()) {
      this.#pruneExpired(s, now);
      total += s.pendingApprovals.size;
    }
    return total;
  }
  #consumeOneTimeGrant(s, fingerprint, now) {
    const grant = s.oneTimeGrants.get(fingerprint);
    if (!grant)
      return false;
    s.oneTimeGrants.delete(fingerprint);
    return grant.expiresAt > now;
  }
  #requireApproval(s, req, now) {
    let existing;
    for (const p of s.pendingApprovals.values()) {
      if (p.fingerprint === req.fingerprint) {
        existing = p;
        break;
      }
    }
    const approval = existing ?? {
      id: `apr_${++this.#seq}_${now}`,
      threadId: s.threadId,
      category: req.category,
      fingerprint: req.fingerprint,
      summary: req.summary,
      url: req.url,
      tabId: req.tabId,
      method: req.method,
      createdAt: now,
      expiresAt: now + APPROVAL_TTL_MS
    };
    if (!existing) {
      s.pendingApprovals.set(approval.id, approval);
    }
    return {
      kind: "require_approval",
      approval,
      errorMessage: `ORYN_APPROVAL_REQUIRED(${approval.id}): 等待用户在 Chrome 扩展中确认授权（${approval.summary}）。请暂停重复调用，等用户批准后再重试同一操作。`
    };
  }
}

var NOTIFICATION_ICON = "data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+X2S8AAAAASUVORK5CYII=";
function normalizeServerUrl(raw) {
  const trimmed = raw.trim().replace(/\/+$/, "");
  if (!trimmed)
    return "";
  if (/^https?:\/\//i.test(trimmed))
    return trimmed;
  if (/^(localhost|127\.0\.0\.1|\[::1\])(?::\d+)?$/i.test(trimmed))
    return `http://${trimmed}`;
  return `https://${trimmed}`;
}
function toWebSocketUrl(baseHttpUrl, pathWithQuery) {
  const u = new URL(pathWithQuery, `${baseHttpUrl}/`);
  u.protocol = u.protocol === "https:" ? "wss:" : "ws:";
  return u.toString();
}
async function loadCloudConfig() {
  const stored = await chrome.storage.local.get({
    mode: "local",
    orynServerUrl: "",
    orynDeviceToken: "",
    orynDeviceId: "",
    orynDeviceName: ""
  });
  return {
    mode: stored.mode === "oryn" ? "oryn" : "local",
    orynServerUrl: typeof stored.orynServerUrl === "string" ? stored.orynServerUrl : "",
    orynDeviceToken: typeof stored.orynDeviceToken === "string" ? stored.orynDeviceToken : "",
    orynDeviceId: typeof stored.orynDeviceId === "string" ? stored.orynDeviceId : "",
    orynDeviceName: typeof stored.orynDeviceName === "string" ? stored.orynDeviceName : ""
  };
}

class OrynCloudClient {
  policy = new CloudPolicyEngine;
  #host;
  #ws = null;
  #pingTimer = null;
  #reconnectTimer = null;
  #reconnectDelay = RECONNECT_MIN_MS;
  #stopped = true;
  #connected = false;
  #lastError = "";
  #config = {
    mode: "local",
    orynServerUrl: "",
    orynDeviceToken: "",
    orynDeviceId: "",
    orynDeviceName: ""
  };
  #pairing = null;
  #threads = [];
  #openSessions = new Map;
  constructor(host) {
    this.#host = host;
  }
  get active() {
    return !this.#stopped && this.#config.mode === "oryn";
  }
  async syncConfig() {
    this.#config = await loadCloudConfig();
  }
  async start() {
    this.#config = await loadCloudConfig();
    if (this.#config.mode !== "oryn") {
      await this.stop();
      return;
    }
    this.#stopped = false;
    await this.connect();
  }
  async stop() {
    this.#stopped = true;
    this.#clearTimers();
    if (this.#ws) {
      const old = this.#ws;
      this.#ws = null;
      old.close();
    }
    this.#connected = false;
    const tabIds = this.policy.clearAll();
    this.#openSessions.clear();
    await Promise.all(tabIds.map((id) => this.#host.detachAndUngroup(id).catch(() => {})));
    await this.#refreshBadge();
  }
  async connect() {
    if (this.#stopped || this.#config.mode !== "oryn")
      return;
    if (this.#ws && (this.#ws.readyState === WebSocket.OPEN || this.#ws.readyState === WebSocket.CONNECTING))
      return;
    const base = normalizeServerUrl(this.#config.orynServerUrl);
    const token = this.#config.orynDeviceToken.trim();
    if (!base || !token) {
      this.#connected = false;
      await this.#refreshBadge();
      return;
    }
    try {
      const ticketResp = await fetch(`${base}/api/v1/ws-ticket`, {
        method: "POST",
        headers: { Authorization: `Bearer ${token}` }
      });
      if (ticketResp.status === 401 || ticketResp.status === 403) {
        await this.#handleRevoked("设备令牌已失效或被吊销，请重新配对");
        return;
      }
      if (!ticketResp.ok) {
        throw new Error(`ws-ticket HTTP ${ticketResp.status}`);
      }
      const { ticket } = await ticketResp.json();
      if (!ticket)
        throw new Error("empty ws-ticket");
      const wsUrl = toWebSocketUrl(base, `/api/v1/browser-relay/ws?ticket=${encodeURIComponent(ticket)}`);
      const socket = new WebSocket(wsUrl);
      this.#ws = socket;
      socket.onopen = () => {
        if (this.#ws !== socket)
          return;
        this.#reconnectDelay = RECONNECT_MIN_MS;
        this.#connected = true;
        this.#lastError = "";
        this.#sendRegister();
        this.#refreshBadge();
        clearInterval(this.#pingTimer ?? undefined);
        this.#pingTimer = setInterval(() => this.#send({ t: "ping" }), CLOUD_PING_MS);
      };
      socket.onmessage = (ev) => {
        if (this.#ws !== socket || typeof ev.data !== "string")
          return;
        this.#handleServerMessage(ev.data);
      };
      socket.onclose = (ev) => {
        if (this.#ws !== socket)
          return;
        this.#ws = null;
        this.#connected = false;
        clearInterval(this.#pingTimer ?? undefined);
        this.#pingTimer = null;
        if (ev.code === CLOSE_REVOKED) {
          this.#handleRevoked("设备已被吊销，请重新配对");
          return;
        }
        this.#refreshBadge();
        if (!this.#stopped)
          this.#scheduleReconnect();
      };
      socket.onerror = () => {
        this.#lastError = "无法连接到 Oryn 服务器";
        socket.close();
      };
    } catch (err) {
      this.#connected = false;
      this.#lastError = err instanceof Error ? err.message : String(err);
      await this.#refreshBadge();
      if (!this.#stopped)
        this.#scheduleReconnect();
    }
  }
  async startPairing(rawServerUrl, deviceName) {
    const serverUrl = normalizeServerUrl(rawServerUrl);
    if (!serverUrl)
      throw new Error("请输入有效的 Oryn 服务器地址");
    const name = deviceName?.trim() || "Chrome Extension";
    const resp = await fetch(`${serverUrl}/api/v1/devices/pair/start`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ name, kind: "browser_extension", platform: "chrome" })
    });
    if (!resp.ok) {
      throw new Error(`配对请求失败（HTTP ${resp.status}）`);
    }
    const data = await resp.json();
    this.#pairing = {
      serverUrl,
      code: data.code,
      verifier: data.verifier,
      verifyUrl: data.verifyUrl,
      expiresAt: data.expiresAt
    };
    await chrome.storage.local.set({ mode: "oryn", orynServerUrl: serverUrl });
    this.#config = await loadCloudConfig();
    if (data.verifyUrl) {
      await chrome.tabs.create({ url: data.verifyUrl, active: true }).catch(() => {});
    }
    return this.#pairing;
  }
  async pollPairingOnce() {
    if (!this.#pairing)
      return { approved: false };
    const p = this.#pairing;
    if (Date.now() >= p.expiresAt) {
      this.#pairing = null;
      return { approved: false, expired: true };
    }
    const resp = await fetch(`${p.serverUrl}/api/v1/devices/pair/poll`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ code: p.code, verifier: p.verifier })
    });
    if (resp.status === 202)
      return { approved: false };
    if (resp.status === 404 || resp.status === 410) {
      this.#pairing = null;
      return { approved: false, expired: true };
    }
    if (!resp.ok) {
      throw new Error(`轮询配对失败（HTTP ${resp.status}）`);
    }
    const data = await resp.json();
    this.#pairing = null;
    await chrome.storage.local.set({
      mode: "oryn",
      orynServerUrl: p.serverUrl,
      orynDeviceToken: data.deviceToken,
      orynDeviceId: data.device.id,
      orynDeviceName: data.device.name
    });
    this.#config = await loadCloudConfig();
    this.#stopped = false;
    await this.connect();
    return { approved: true };
  }
  async unpair() {
    this.#pairing = null;
    await this.stop();
    await chrome.storage.local.set({
      orynDeviceToken: "",
      orynDeviceId: "",
      orynDeviceName: ""
    });
    this.#config = await loadCloudConfig();
    await this.#refreshBadge();
  }

    }

  onTabEvent(ev) {
    if (!this.active || !this.#connected)
      return;
    const tabId = ev.t === "tabRemoved" ? ev.tabId : ev.tab.tabId;
    for (const threadId of this.#openSessions.keys()) {
      const filtered = this.policy.filterTabEvent(threadId, ev);
      if (!filtered)
        continue;
      if (ev.t === "tabRemoved") {
        this.policy.unshareTab(threadId, tabId);
        this.#sendSessionState(threadId);
      }
      this.#sendFrame(threadId, filtered);
    }
  }

        this.#sendFrame(threadId, ev);
      }
    }

  getPopupSnapshot() {

    for (const tid of this.#openSessions.keys()) {
      const sum = this.policy.getSummary(tid);
      sessions[tid] = sum;
      pendingApprovals.push(...sum.pendingApprovals);
    }

      serverUrl: this.#config.orynServerUrl,
      paired: Boolean(this.#config.orynDeviceToken),
      deviceId: this.#config.orynDeviceId,
      deviceName: this.#config.orynDeviceName,
      connected: this.#connected,
      lastError: this.#lastError,
      pairing: this.#pairing,

      pendingApprovals

  }
  async#sendRegister() {
    const hello = await this.#host.buildFullHello();
    this.#send({
      t: "register",
      instanceId: hello.instanceId ?? "",
      browserSession: hello.browserSession ?? "",
      browserVersion: hello.browserVersion,
      userAgent: hello.userAgent,
      extVersion: hello.ompVersion ?? "0.2.0"
    });
  }
  async#handleServerMessage(raw) {
    let msg;
    try {
      msg = JSON.parse(raw);
    } catch {
      return;
    }
    switch (msg.t) {
      case "pong":
        return;
      case "registered":
        if (Array.isArray(msg.threads))
          this.#threads = msg.threads;
        this.#notifyPopup();
        return;
      case "threads.update": {
        if (Array.isArray(msg.threads)) {
          const changed = JSON.stringify(this.#threads) !== JSON.stringify(msg.threads);
          this.#threads = msg.threads;
          if (changed)
            this.#notifyPopup();
        }
        return;
      }
      case "session.open": {
        if (!msg.threadId)
          return;
        this.#openSessions.set(msg.threadId, {
          threadTitle: msg.threadTitle ?? msg.threadId,
          projectName: msg.projectName ?? ""
        });
        const fullHello = await this.#host.buildFullHello();
        const filtered = this.policy.filterHello(msg.threadId, fullHello);
        this.#sendFrame(msg.threadId, filtered);
        this.#sendSessionState(msg.threadId);
        await this.#refreshBadge();
        this.#notifyPopup();
        return;
      }
      case "session.close": {
        if (!msg.threadId)
          return;
        this.#openSessions.delete(msg.threadId);
        if (msg.reason === "unbound" || msg.reason === "revoked" || msg.reason === "rebound") {
          const tabs = this.policy.clearSession(msg.threadId);
          await Promise.all(tabs.map((id) => this.#host.detachAndUngroup(id).catch(() => {})));
        }
        await this.#refreshBadge();
        this.#notifyPopup();
        return;
      }
      case "session.frame": {
        if (!msg.threadId || !msg.payload)
          return;
        await this.#handleRelayPayload(msg.threadId, msg.payload);
        return;
      }
    }
  }
  async#handleRelayPayload(threadId, payload) {
    if (payload.t === "reload" || payload.t === "pong")
      return;
    const decision = this.policy.authorizeRpc(threadId, payload);
    if (decision.kind === "deny") {
      this.#sendFrame(threadId, { t: "rpcResult", id: payload.id, ok: false, error: decision.reason });
      return;
    }
    if (decision.kind === "require_approval") {
      this.#sendFrame(threadId, { t: "rpcResult", id: payload.id, ok: false, error: decision.errorMessage });
      this.#sendSessionState(threadId);
      await this.#refreshBadge();
      this.#notifyPopup();
      await this.#notifyApproval(threadId, decision.approval);
      return;
    }
    try {
      const result = await this.#host.executeRpc(payload);
      if (payload.op === "createTab" && result && typeof result === "object" && "tab" in result) {
        const maybeTab = result.tab;
        if (maybeTab && typeof maybeTab === "object" && "tabId" in maybeTab && typeof maybeTab.tabId === "number") {
          this.policy.registerCreatedTab(threadId, maybeTab);
          this.#sendSessionState(threadId);
        }
      }
      this.#sendFrame(threadId, { t: "rpcResult", id: payload.id, ok: true, result });
    } catch (err) {
      this.#sendFrame(threadId, {
        t: "rpcResult",
        id: payload.id,
        ok: false,
        error: err instanceof Error ? err.message : String(err)
      });
    }
  }
  async#notifyApproval(threadId, approval) {
    const meta = this.#openSessions.get(threadId);
    const title = `Oryn 授权请求（${meta?.threadTitle ?? threadId}）`;
    try {
      await chrome.notifications?.create(approval.id, {
        type: "basic",
        iconUrl: NOTIFICATION_ICON,
        title,
        message: `${approval.summary} — 点击浏览器右上角 Oryn 扩展图标确认`,
        priority: 2
      });
    } catch {}
  }
  #notifyPopup() {
    chrome.runtime.sendMessage({ t: "cloud.changed" }).catch(() => {});
  }
  async#handleRevoked(reason) {
    this.#lastError = reason;
    await this.unpair();
  }
  #sendSessionState(threadId) {
    const sum = this.policy.getSummary(threadId);
    this.#send({
      t: "session.state",
      threadId,
      sharedTabCount: sum.sharedTabs.length,
      pendingApprovals: sum.pendingApprovals.length,
      autoApproveCreateTab: sum.autoApproveCreateTab
    });
  }
  #sendFrame(threadId, payload) {
    this.#send({ t: "session.frame", threadId, payload });
  }
  #send(msg) {
    if (this.#ws && this.#ws.readyState === WebSocket.OPEN) {
      this.#ws.send(JSON.stringify(msg));
    }
  }
  async#refreshBadge() {
    const pending = this.policy.totalPendingApprovals();
    if (pending > 0) {
      await this.#host.setBadge("!", "#d97706");
      return;
    }
    if (this.#connected) {
      await this.#host.setBadge("on", "#1a7f37");
      return;
    }
    await this.#host.setBadge("off", "#8b8b8b");
  }
  #scheduleReconnect() {
    if (this.#reconnectTimer)
      return;
    const delay = this.#reconnectDelay;
    this.#reconnectDelay = Math.min(this.#reconnectDelay * 2, RECONNECT_MAX_MS);
    this.#reconnectTimer = setTimeout(() => {
      this.#reconnectTimer = null;
      this.connect();
    }, delay);
  }
  #clearTimers() {
    clearInterval(this.#pingTimer ?? undefined);
    this.#pingTimer = null;
    clearTimeout(this.#reconnectTimer ?? undefined);
    this.#reconnectTimer = null;
  }
}

async function ensureInstanceId() {
  if (instanceId)
    return instanceId;
  const key = "relayInstanceId";
  const stored = await chrome.storage.local.get({ [key]: "" });
  const existing = stored[key];
  instanceId = typeof existing === "string" && existing.length > 0 ? existing : crypto.randomUUID();
  await chrome.storage.local.set({ [key]: instanceId });
  return instanceId;
}
var browserSession = null;
function ensureBrowserSession() {
  browserSession ??= (async () => {
    const key = "relayBrowserSession";
    const stored = await chrome.storage.session.get({ [key]: "" });
    const existing = stored[key];
    if (typeof existing === "string" && existing.length > 0)
      return existing;
    const created = crypto.randomUUID();
    await chrome.storage.session.set({ [key]: created });
    return created;
  })();
  return browserSession;
}

function snapshot(tab) {
  if (tab.id === undefined)
    return null;
  return {
    tabId: tab.id,
    url: tab.url ?? tab.pendingUrl ?? "",
    title: tab.title ?? "",
    active: tab.active,
    discarded: tab.discarded === true,
    windowId: tab.windowId,
    pinned: tab.pinned,
    groupId: tab.groupId
  };
}
var ompGroupTitle = null;
var groupOps = Promise.resolve();
function enqueueGroupOp(fn) {
  const result = groupOps.then(fn, fn);
  groupOps = result.catch(() => {});
  return result;
}
async function groupTabs(tabIds, title, color) {
  ompGroupTitle = title;
  chrome.storage.session.set({ ompGroupTitle: title });
  const byWindow = new Map;
  for (const tabId of tabIds) {
    try {
      const tab = await chrome.tabs.get(tabId);
      if (tab.pinned || tab.id === undefined)
        continue;
      const bucket = byWindow.get(tab.windowId) ?? [];
      bucket.push(tab.id);
      byWindow.set(tab.windowId, bucket);
    } catch {}
  }
  const grouped = {};
  for (const [windowId, ids] of byWindow) {
    const existing = await chrome.tabGroups.query({ title, windowId });
    let groupId;
    if (existing[0]) {
      groupId = existing[0].id;
      for (const dupe of existing.slice(1)) {
        const dupeTabs = await chrome.tabs.query({ groupId: dupe.id });
        const dupeIds = dupeTabs.map((tab) => tab.id).filter((id) => id !== undefined);
        if (dupeIds.length > 0)
          await chrome.tabs.group({ tabIds: dupeIds, groupId });
      }
      await chrome.tabs.group({ tabIds: ids, groupId });
    } else {
      groupId = await chrome.tabs.group({ tabIds: ids });
    }
    await chrome.tabGroups.update(groupId, { title, color });
    for (const id of ids)
      grouped[String(id)] = groupId;
  }
  return { grouped };
}
async function restoreGroups() {
  if (!ompGroupTitle) {
    const stored = await chrome.storage.session.get({ ompGroupTitle: "" }).catch(() => ({ ompGroupTitle: "" }));
    ompGroupTitle = typeof stored.ompGroupTitle === "string" && stored.ompGroupTitle ? stored.ompGroupTitle : null;
  }
  if (!ompGroupTitle)
    return;
  const groups = await chrome.tabGroups.query({ title: ompGroupTitle }).catch(() => []);
  for (const group of groups) {
    const tabs = await chrome.tabs.query({ groupId: group.id }).catch(() => []);
    const ids = tabs.map((tab) => tab.id).filter((id) => id !== undefined);
    if (ids.length > 0)
      await chrome.tabs.ungroup(ids).catch(() => {});
  }
}

async function removeTab(tabId) {
  for (let attempt = 1;; attempt++) {
    try {
      await chrome.tabs.remove(tabId);
      return;
    } catch (error) {
      if (/No tab with id/.test(String(error)))
        return;
      if (attempt >= REMOVE_TAB_ATTEMPTS)
        throw error;
      const pause = Promise.withResolvers();
      setTimeout(pause.resolve, REMOVE_TAB_RETRY_MS);
      await pause.promise;
    }
  }
}

async function buildHello() {
  const [tabs, targets] = await Promise.all([chrome.tabs.query({}), chrome.debugger.getTargets()]);
  const snapshots = [];
  for (const tab of tabs) {
    const snap = snapshot(tab);
    if (snap)
      snapshots.push(snap);
  }
  const attachedTabIds = await ownedDebuggerTabs(targets, (tabId) => chrome.debugger.sendCommand({ tabId }, "Target.getTargetInfo"));
  const versionMatch = /Chrome\/[\d.]+/.exec(navigator.userAgent);
  return {
    t: "hello",
    instanceId: await ensureInstanceId(),
    browserSession: await ensureBrowserSession(),
    userAgent: navigator.userAgent,
    browserVersion: versionMatch?.[0] ?? "Chrome/unknown",
    discardedTabsProtocol: 1,
    ompVersion: typeof OMP_INSTALL_VERSION === "string" ? OMP_INSTALL_VERSION : undefined,
    reloadProtocol: EXTENSION_RELOAD_PROTOCOL_VERSION,
    downloadsProtocol: chrome.downloads === undefined ? undefined : DOWNLOADS_PROTOCOL_VERSION,
    tabs: snapshots,
    attachedTabIds
  };
}
async function runRpc(msg, origin) {
  switch (msg.op) {
    case "attach": {
      const attach = () => chrome.debugger.attach({ tabId: msg.tabId }, "1.3");
      await (msg.clearForeignFrames ? pastForeignFrames(msg.tabId, attach) : attach());
      return {};
    }
    case "detach":
      relayInitiatedDetachTabs.add(msg.tabId);
      try {
        await chrome.debugger.detach({ tabId: msg.tabId });
        return {};
      } catch (error) {
        relayInitiatedDetachTabs.delete(msg.tabId);
        throw error;
      }
    case "send": {
      const send = () => chrome.debugger.sendCommand(msg.sessionId ? { tabId: msg.tabId, sessionId: msg.sessionId } : { tabId: msg.tabId }, msg.method, msg.params);
      return await (msg.clearForeignFrames ? pastForeignFrames(msg.tabId, send) : send());
    }
    case "createTab": {
      const tab = await chrome.tabs.create({ url: msg.url, active: msg.active ?? true });
      const snap = snapshot(tab);
      if (!snap)
        throw new Error("created tab has no id");
      if (origin && origin.readyState !== WebSocket.OPEN)
        removeTab(snap.tabId).catch(() => {});
      return { tab: snap };
    }
    case "removeTab":
      await removeTab(msg.tabId);
      return {};
    case "activateTab": {
      const tab = await chrome.tabs.get(msg.tabId);
      await chrome.windows.update(tab.windowId, { focused: true });
      await chrome.tabs.update(msg.tabId, { active: true });
      return {};
    }
    case "group":
      return await enqueueGroupOp(() => groupTabs(msg.tabIds, msg.title, msg.color));
    case "ungroup":
      await enqueueGroupOp(() => chrome.tabs.ungroup(msg.tabIds).catch(() => {}));
      return {};
    case "findDownload": {
      if (chrome.downloads === undefined)
        return null;
      const id = downloads.idFor(msg.guid);
      const [item] = id === undefined ? [] : await chrome.downloads.search({ id });
      if (item && item.state !== "in_progress")
        downloads.forget(msg.guid);
      const found = item ? { filename: item.filename, state: item.state } : null;
      return found;
    }
  }
}
function handleRelayMessage(socket, raw) {
  let msg;
  try {
    msg = JSON.parse(raw);
  } catch {
    return;
  }
  if (msg.t === "pong")
    return;
  if (msg.t === "reload") {
    chrome.runtime.reload();
    return;
  }
  const answer = (reply) => {
    if (socket.readyState === WebSocket.OPEN)
      socket.send(JSON.stringify(reply));
  };
  runRpc(msg, socket).then((result) => answer({ t: "rpcResult", id: msg.id, ok: true, result })).catch((err) => {
    answer({ t: "rpcResult", id: msg.id, ok: false, error: err instanceof Error ? err.message : String(err) });
  });
}
function scheduleReconnect() {
  const delay = reconnectDelay;
  reconnectDelay = Math.min(reconnectDelay * 2, RECONNECT_MAX_MS2);
  setTimeout(() => void connect(), delay);
}

    }

});
async function connect() {
  const cloudCfg = await loadCloudConfig();
  if (cloudCfg.mode === "oryn") {
    if (ws) {
      const old = ws;
      ws = null;
      old.close();
    }
    clearInterval(pingTimer ?? undefined);
    pingTimer = null;
    await cloud.start();
    return;
  }
  await cloud.stop();
  if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING))
    return;
  const settings = await loadSettings();
  const url = `ws://127.0.0.1:${settings.port}/ext${settings.token ? `?token=${encodeURIComponent(settings.token)}` : ""}`;
  const socket = new WebSocket(url);
  ws = socket;
  socket.onopen = () => {
    reconnectDelay = RECONNECT_MIN_MS2;
    setBadge(true);
    buildHello().then((hello) => post(hello));
    clearInterval(pingTimer ?? undefined);
    pingTimer = setInterval(() => post({ t: "ping" }), PING_INTERVAL_MS);
  };
  socket.onmessage = (event) => {
    if (typeof event.data === "string")
      handleRelayMessage(socket, event.data);
  };
  socket.onclose = () => {
    if (ws !== socket)
      return;
    ws = null;
    clearInterval(pingTimer ?? undefined);
    pingTimer = null;
    setBadge(false);
    restoreGroups();
    scheduleReconnect();
  };
  socket.onerror = () => {
    socket.close();
  };
}

chrome.runtime.onMessage.addListener((raw, _sender, sendResponse) => {
  if (!raw || typeof raw !== "object" || !("t" in raw) || typeof raw.t !== "string")
    return;
  const msg = raw;
  const handle = async () => {
    switch (msg.t) {
      case "popup.getState": {
        await cloud.syncConfig();
        const isShareableUrl = (u) => /^https?:\/\//i.test(u);
        const [activeTab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true });
        let candidate = activeTab;
        if (!candidate || candidate.id === undefined || !isShareableUrl(candidate.url ?? candidate.pendingUrl ?? "")) {
          const allTabs = await chrome.tabs.query({});
          candidate = allTabs.find((t) => t.active && t.id !== undefined && isShareableUrl(t.url ?? t.pendingUrl ?? "")) ?? allTabs.find((t) => t.id !== undefined && isShareableUrl(t.url ?? t.pendingUrl ?? ""));
        }
        const currentTab = candidate && candidate.id !== undefined ? { tabId: candidate.id, title: candidate.title ?? "", url: candidate.url ?? candidate.pendingUrl ?? "" } : null;
        return {
          cloud: cloud.getPopupSnapshot(),
          localConnected: ws?.readyState === WebSocket.OPEN,
          currentTab
        };
      }
      case "popup.setMode": {
        const nextMode = msg.mode === "oryn" ? "oryn" : "local";
        await chrome.storage.local.set({ mode: nextMode });
        await connect();
        return { ok: true };
      }
      case "popup.startPairing":
        try {
          const pairing = await cloud.startPairing(String(msg.serverUrl ?? ""), String(msg.deviceName ?? ""));
          return { ok: true, pairing };
        } catch (err) {
          return { ok: false, error: err instanceof Error ? err.message : String(err) };
        }
      case "popup.pollPairing":
        return await cloud.pollPairingOnce();
      case "popup.unpair":
        await cloud.unpair();
        return { ok: true };
      case "popup.bindThread":
        await cloud.bindThread(String(msg.threadId ?? ""));
        return { ok: true };
      case "popup.unbindThread":
        await cloud.unbindThread(String(msg.threadId ?? ""));
        return { ok: true };
      case "popup.shareTab":
        await cloud.shareTab(String(msg.threadId ?? ""), Number(msg.tabId));
        return { ok: true };
      case "popup.unshareTab":
        await cloud.unshareTab(String(msg.threadId ?? ""), Number(msg.tabId));
        return { ok: true };
      case "popup.resolveApproval":
        await cloud.resolveApproval(String(msg.threadId ?? ""), String(msg.approvalId ?? ""), msg.decision);
        return { ok: true };
      case "popup.emergencyDisconnect":
        await cloud.emergencyDisconnectAll();
        return { ok: true };
      case "popup.refreshThreads":
        cloud.requestThreadsRefresh();
        return { ok: true };
      default:
        return { ok: false, error: "unknown message" };
    }
  };
  handle().then(sendResponse);
  return true;
});
